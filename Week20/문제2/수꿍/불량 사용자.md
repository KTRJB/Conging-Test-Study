```swift
import Foundation

// 주어진 userID와 bannedID에 따라 가능한 제재 아이디 목록의 경우 수를 계산하는 함수
func programmers_week20_2(_ userID: [String], _ bannedID: [String]) -> Int {
    // 가능한 제재 아이디 목록을 저장할 집합
    var result = Set<Set<String>>()
    // 현재 선택된 제재 아이디 목록
    var currentSet = Set<String>()

    // 재귀 함수 호출하여 제재 아이디 조합을 찾기 시작
    findBannedCombinations(userID, bannedID, 0, &currentSet, &result)

    // 중복된 목록은 Set으로 관리되므로 결과의 크기만 반환
    return result.count
}

// 주어진 userID와 bannedID를 기반으로 가능한 조합을 찾는 재귀 함수
private func findBannedCombinations(
    _ userID: [String],
    _ bannedID: [String],
    _ index: Int,
    _ currentSet: inout Set<String>,
    _ result: inout Set<Set<String>>
) {
    // 모든 banned_id를 처리한 경우, 결과에 현재 선택된 집합을 추가
    if index == bannedID.count {
        result.insert(currentSet)
        return
    }

    // userID 목록을 순회
    for user in userID {
        // 이미 사용된 아이디는 패스
        if currentSet.contains(user) {
            continue
        }

        // 현재 사용자 아이디가 bannedID의 패턴과 일치하는지 확인
        if isMatching(user, bannedID[index]) {
            // 일치하는 경우 현재 선택된 제재 아이디 목록에 추가
            currentSet.insert(user)
            // 다음 bannedID로 넘어가기 위해 재귀 호출
            findBannedCombinations(userID, bannedID, index + 1, &currentSet, &result)
            // 백트래킹을 위해 현재 아이디 제거
            currentSet.remove(user)
        }
    }
}

// 주어진 userID가 bannedID의 패턴과 일치하는지 확인하는 함수
private func isMatching(_ userID: String, _ bannedID: String) -> Bool {
    // 길이가 다르면 일치하지 않음
    if userID.count != bannedID.count {
        return false
    }

    // 각 문자 비교
    for (userChr, bannedChr) in zip(userID, bannedID) {
        // bannedID에서 '*'가 아닌 경우, 문자가 일치하지 않으면 false
        if bannedChr != "*" && userChr != bannedChr {
            return false
        }
    }

    // 모든 문자 비교를 통과하면 true 반환
    return true
}
```
