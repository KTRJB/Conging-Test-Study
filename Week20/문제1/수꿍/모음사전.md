func programmers_week20_1(_ word: String) -> Int {
    let vowels = ["A", "E", "I", "O", "U"]  // 모음 배열
    // 각 자리에 대한 가중치
    // 5^4 + 5^3 + 5^2 + 5^1 + 5^0
    // 5^3 + 5^2 + 5^1 + 5^0
    // 5^2 + 5^1 + 5^0
    // 5^1 + 5^0
    // 5^0
    var multiplier = 781 // 각 자리의 가중치를 미리 계산해 둠 (5^4 + 5^3 + 5^2 + 5^1 + 5^0)
    var answer = 0       // 단어의 위치

    // 각 문자의 위치에 따라 가중치를 더해 나감
    for char in word {
        if let index = vowels.firstIndex(of: String(char)) {
            // 해당 자리에 위치한 모음이 몇 번째인지 계산하고 가중치를 곱해서 더해줌
            answer += index * multiplier + 1
        }
        // 다음 자리는 가중치가 5배 줄어듦
        multiplier = (multiplier - 1) / 5
    }

    return answer
}
